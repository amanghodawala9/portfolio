x = np.linspace(-3, 3, 250)

x = x.reshape(-1, )
y = y.reshape(-1, )
In [4]:
transform_matrix = np.array([[1,1],
                             [0,1]])

transformed = transform_matrix @ np.array([x,y]) # applying the transformation on each pair of (x,y) in the grid

new_x, new_y = transformed[0], transformed[1] # separating x and y

plt.figure(figsize=(10, 6))
plt.axis('equal')
plt.scatter(x, y, s=5, alpha=0.8)
plt.scatter(new_x, new_y, c='r', s=10, alpha=0.1)
